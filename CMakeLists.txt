cmake_minimum_required(VERSION 3.16)
project(cybr_instrument_bakers LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP QUIET)
foreach(baker bake_lighting bake_irradiance)
  add_executable(${baker} instrument-rebuild/studio/${baker}.cpp)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(${baker} PRIVATE OpenMP::OpenMP_CXX)
  endif()
endforeach()
if(NOT OpenMP_CXX_FOUND)
  message(STATUS "OpenMP unavailable: bakers compile with serial loops; sampling is unchanged.")
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME bounded_bakers COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tools/test_bakers.py
    $<TARGET_FILE:bake_lighting> $<TARGET_FILE:bake_irradiance>)
endif()
